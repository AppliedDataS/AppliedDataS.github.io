def convert_degrees2coordinates(degrees_string):
  """
  The convert_de function takes a string of the form 'dddmm.mmmmN' or 'dddmm.mmmmW' and returns
  a string of the form '-ddd.mmmmmmm'.  This is useful for converting coordinates in degrees, minutes, and decimal
  minutes to decimal degrees.
  Parameters
  ----------
      rees2coordinates(degrees_string
          Convert the string of degrees into a float
  Returns
  -------
      A string that is a number with a minus sign if the coordinate is west or south
  Doc Author
  ----------
      Trelent
  """
  if degrees_string.strip()[-1]=='W':
    return (f'-{degrees_string.strip()[:-2]}')
  else:
    return (f'{degrees_string.strip()[:-2]}')

def get_stadiums():
  """
                   The get_stadiums function returns a list of stadiums in the database.
                   Parameters
                   ----------
                   Returns
                   -------
                       A list of stadiums
                   Doc Author
                   ----------
                       Trelent
                   """
  stadiums = pd.read_html('https://en.wikipedia.org/wiki/List_of_Premier_League_stadiums')[0]
  stadiums['Coordinates'] = stadiums['Coordinates'].str.split(
    '/').str[-1].str.strip().str.split(
      ' ').apply(lambda stringCors:[elm.split()[-1] for elm in stringCors])
  stadiums['lat'] = pd.to_numeric(stadiums['Coordinates'].apply(
    lambda lst: lst[0]).apply(convert_degrees2coordinates).str.encode(
      'ascii', 'ignore').str.decode('ascii'),errors='coerce')
  stadiums['long'] = pd.to_numeric(stadiums['Coordinates'].apply(
    lambda lst: lst[-1]).apply(convert_degrees2coordinates).str.encode(
      'ascii', 'ignore').str.decode('ascii'),errors='coerce')
  return stadiums

def create_url(params):
    """
    The create_url function takes a dictionary of parameters and returns a url string.
    The function is used to create the url for the API call.
    Parameters
    ----------
        params
            Pass the parameters to the api
    Returns
    -------
        A string that is the url for a query to openmeteodata
    Doc Author
    ----------
        Trelent
    """
    url = "https://archive-api.open-meteo.com/v1/archive?"
    for k, v in params.items():
        if type(v) == list:  # noqa: E721
            url += f"{k}={','.join(v)}"
        else:
            url += f"{k}={v}"
        url += "&"
    return url[:-1]
def get_weather(latitude, longitude, start_date, end_date):
    """
    The get_weather function takes in a latitude, longitude, start_date and end_date.
    It then creates a url using the create_url function with the parameters passed into get_weather.
    The url is then used to make an API call to Weatherbit's weather API and returns hourly and daily dataframes.
    Parameters
    ----------
        latitude
            Specify the latitude of the location for which you want to get weather data
        longitude
            Specify the longitude of the location
        start_date
            Define the start date of the weather data to be retrieved
        end_date
            Specify the end date of the data to be retrieved
    Returns
    -------
        A tuple of two dataframes
    Doc Author
    ----------
        Trelent
    """
    params = {
        "latitude": latitude,
        "longitude": longitude,
        "start_date": start_date,
        "end_date": end_date,
        "temperature_unit":'celsius',
        "hourly": ["temperature_2m", "relative_humidity_2m", "dew_point_2m", "apparent_temperature", "precipitation", "rain", "snowfall", "snow_depth", "weather_code", "pressure_msl", "surface_pressure", "cloud_cover", "cloud_cover_low", "cloud_cover_mid", "cloud_cover_high", "et0_fao_evapotranspiration", "vapour_pressure_deficit", "wind_speed_10m", "wind_speed_100m", "wind_direction_10m", "wind_direction_100m", "wind_gusts_10m", "soil_temperature_0_to_7cm"],
	"daily": ["temperature_2m_max", "temperature_2m_min", "temperature_2m_mean", "apparent_temperature_max", "apparent_temperature_min", "apparent_temperature_mean", "sunrise", "sunset", "rain_sum", "snowfall_sum", "precipitation_hours", "wind_speed_10m_max", "wind_gusts_10m_max"]
}
    url = create_url(params)
    try:
        r = requests.get(url).json()

        hourly = pd.DataFrame(r["hourly"], columns=r["hourly_units"])
        daily = pd.DataFrame(r["daily"], columns=r["daily_units"])
        return hourly, daily
    except Exception as e:
        return

def combine(row):
  # print(row['home_team'])
  """
  The combine function takes a row from the fixtures dataframe and merges it with the hourly weather data.
  It returns a new dataframe containing all of this information.
  Parameters
  ----------
      row
          Pass a row from the dataframe to the function
  Returns
  -------
      A dataframe
  Doc Author
  ----------
      Trelent
  """
  hourley = pd.read_csv(row['hourly'], 
                        index_col=[0],
                        )
  hourley['time'] = pd.to_datetime(hourley['time'],errors='coerce').dt.tz_localize('UTC')
  tmp = fixtures[fixtures['home_team']==row['name']]
  tmp = tmp.merge(hourley,left_on='kickoff_time',right_on='time',how='left')
  tmp = tmp.merge(hourley,left_on='end_time',right_on='time',how='left')
  return tmp
